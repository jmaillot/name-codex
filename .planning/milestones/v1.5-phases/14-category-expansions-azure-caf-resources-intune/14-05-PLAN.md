    In src/lib/segments.ts change fieldTip to accept an optional convention id and prefer the rule-scoped key:

    ```typescript
    export function fieldTip(field: NamingField | undefined, fallback: string, conventionId?: string): string {
      if (!field?.tip) return fallback;
      if (conventionId) {
        const scoped = tl(`data.rule.${conventionId}.field.${field.name}.tip`, "");
        if (scoped) return scoped;
      }
      return tl(`data.field.${field.name}.tip`, fallback);
    }
    ```

    Lookup precedence: rule-scoped key → legacy global key → inline fallback. This preserves every existing global tip (Region, Environment, GroupName, SiteName, etc.) while giving rule-specific content first claim.

    In src/components/SegmentEditor.tsx line 105 pass the convention id:
    ```tsx
    {field?.tip && <InfoIcon text={fieldTip(field, field.tip, convention.id)} />}
    ```
